\begin{frame}
  \frametitle{Least Squares Regression}
  \begin{displaymath}
    \begin{array}{rcl}
      e & = & y - \hat{y} \\[2em]
      & & e_1,\ e_2,\ \ldots,\ e_n \\[2em]
      & & e_1^2 + e_2^2 + \ldots + e_n^2 
    \end{array}
  \end{displaymath} 
\end{frame}

\begin{frame}
  \frametitle{Split Scatterplot}
  \centering
    \begin{tikzpicture}
    \begin{axis}[
      legend pos=south east,
      xlabel={Total Length (cm)},
      ylabel={Head Length (mm)},
      ]
      \addplot [
      scatter/classes={
        m={mark=square*,red},
        f={mark=square*,blue}
      },
      scatter, 
      only marks,
      scatter src=explicit symbolic,
      ]   
      table [x=total_l,y=head_l,col sep=tab,meta=sex] {../data/possum.dat};
      \legend{Male, Female}
      \addplot+ [
        blue,
        mark=|
        sharp plot,
      ]
      coordinates { (74,81) (98,100) };
      \addplot+ [
        red,
        mark=|
        sharp plot,
      ]
      coordinates { (74,82.66) (98,101.66) };
    \end{axis}
    \end{tikzpicture}
\end{frame}

\begin{frame}

      \text{female} & \mapsto & 1
