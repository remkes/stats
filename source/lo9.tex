\begin{frame}
  \frametitle{Least Squares Regression}
  \begin{displaymath}
    \begin{array}{rcl}
      e & = & y - \hat{y} \\[2em]
      & & e_1,\ e_2,\ \ldots,\ e_n \\[2em]
      & & e_1^2 + e_2^2 + \ldots + e_n^2 
    \end{array}
  \end{displaymath} 
\end{frame}
