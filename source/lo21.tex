      Conducting hypothesis tests and constructing confidence

      interpreting the results.
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Difference of Means}
  \begin{itemize}
    \item[] 
      One numeric variable.
      \vspace{0.5cm}
    \item[] 
      One categorical variable with two groups. 
      \vspace{0.5cm}
    \item[] 
      Independent: the groups are not connected. 
      \vspace{0.5cm}
    \item[] 
      Independent: do babies born to mothers who smoke have a
      different average birthweight than babies botn to mother who do
      not smoke?
      \vspace{0.5cm}
    \item[] 
      Dependent: Is average blood pressure lower afer meditation than
      before meditation?
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Hypothesis testing}
  \begin{itemize}
    \item[] 
      $x = \hat{\mu}_n - \hat{\mu}_s$. 
      \vspace{0.2cm}
    \item[] 
      T-Distribution
      \vspace{0.2cm}
    \item[] 
      $\mu = 0$
      \vspace{0.2cm}
    \item[] 
      $SE = \sqrt{ \frac{\sigma_1^2}{n_1} + \frac{\sigma_2^2}{n_2} }$
      \vspace{0.2cm}
    \item[] 
      Degrees of freedom
      \vspace{0.2cm}
    \item[] 
      $T = \frac{x - 0}{SE}$
      \vspace{0.2cm}
    \item[] 
      p-value
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Confidence Intervals}
  \begin{itemize}
    \item[] 
      mean $x = \hat{\mu}_n - \hat{\mu}_s$
      \vspace{0.5cm}
    \item[] 
      $SE = \sqrt{ \frac{\sigma_1^2}{n_1} + \frac{\sigma_2^2}{n_2} }$
      \vspace{0.5cm}
    \item[] 
      Degrees of freedom
      \vspace{0.5cm}
    \item[] 
      Coefficient $t$
      \vspace{0.5cm}
    \item[] 
      $[x - t(SE), x + t(SE)]$
  \end{itemize}
\end{frame}
