\begin{frame}
  \frametitle{More Vocabulary}
  \begin{itemize}
    \item[] What is the average mercury content in Atlantic Ocean
      swordfish?
      \vspace{0.5cm}
    \item[] Population: all the Atlantic Ocean swordfish.
      \vspace{0.5cm}
    \item[] Sample: a piece of this population, some of the swordfish. 
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Even More Vocabulary}
  \begin{itemize}
    \item[] Population: all possible cases. 
      \vspace{0.5cm}
    \item[] Sample: a piece of this population, some of the cases.
      \vspace{0.5cm}
    \item[] Measurement of a population: parameter. 
      \vspace{0.5cm}
    \item[] Measurement of a sample: statistic. 
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{The Goal!}
  \begin{itemize}
    \item[] We want good statistics, one that match the parameter.
      \vspace{0.5cm}
    \item[] Good statistics need good samples.
      \vspace{0.5cm}
    \item[] Samples should be representative. 
      \vspace{0.5cm}
    \item[] Samples should be unbiased. 
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Anecdotal Evidence}
  \begin{itemize}
    \item[] More cases $\rightarrow$ better statistics. 
      \vspace{0.5cm}
    \item[] Fewer cases $\rightarrow$ worse statistics.
      \vspace{0.5cm}
    \item[] Anecdotal evidence: using only one case or only a few
      sensational cases. The worst statistics. 
      \vspace{0.5cm}
    \item[] Train your mind to resist conclusions based on anecdotal
      evidence. 
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Summary}
  \begin{itemize}
    \item[] We study samples and measure statistics. 
      \vspace{0.5cm}
    \item[] Good experimental design $\rightarrow$ good samples
      $\rightarrow$ good statistics $\rightarrow$ reasonable
      understanding of parameters.
      \vspace{0.5cm}
    \item[] Search for biases and confounding variables. 
  \end{itemize}
