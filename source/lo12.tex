\begin{frame}
  \frametitle{Two Tests for Inference}
  \begin{itemize}
    \item[] 
      Hypothesis Testing
      \vspace{0.5cm}
    \item[] 
      Does this explanatory variable have an effect on this dependent
      variable.
      \vspace{0.5cm}
    \item[] 
      Is this population parameter less than or greater than some
      fixed value?
      \vspace{0.5cm}
    \item[] 
      Calculate probability of randomly generated the sample
      statistics with the null hypothesis.
      \vspace{0.5cm}
    \item[] 
      Confidence Interval
      \vspace{0.5cm}
    \item[] 
      Measure the size of the effect of this explanatory variable on a
      dependent variable.
      \vspace{0.5cm}
    \item[] 
      Measure a range for a population parameter 
      \vspace{0.5cm}
    \item[] 
      Calculate the probability of the population parameter falling
      within a certain interval. 
  \end{itemize}
\end{frame}

    \item[]

\begin{frame}
  \frametitle{Bootstrapping}
  \begin{itemize}
    \item[] 
      Assume the sample statistic \emph{is} the population parameter.
      \vspace{0.5cm}
    \item[] 
      With that assumption, re-draw the sample.
      \vspace{0.5cm}
    \item[] 
      Calculate the probability, either mathematically or by
      simulation.
      \vspace{0.5cm}
    \item[] 
  \end{itemize}
\end{frame}
